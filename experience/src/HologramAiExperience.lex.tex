  result: String,

definition{
  name: offlineReturn,
  parameters: [],
  result: String,
  body: "A second visit needs no network. The files a load read are kept in the browser under their BLAKE3 address, each checked again as it is read back, so a return visit loads with the network off. When the browser is short of room, a kept file is dropped only on the kernel's word."
}

definition{
  name: storedOnce,
  parameters: [],
  result: String,
  body: "A tensor is stored once. Every tensor of every open model has one address, the BLAKE3 of its own bytes, so a tensor already held, under any model, precision or repository, is not downloaded or stored again, and any listed source may serve one, because the address checks the bytes. The bytes and the load time this saves are measured, and so is how many tensors the supported models share."
}

definition{

  result: String,
  body: "The same model, prompt and settings give the same tokens on every GPU, CPU, browser and phone, bit for bit. The engine computes in integers only, with no floating-point operation between the bytes it admits and the token ids it emits."
}

definition{
