\begin{lexlean}{Engine.GpuRun}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Engine.Gpu}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: exec,
  parameters: [
    fuel : Nat,
    ev : (Fn (Engine.Gpu::Machine Engine.Gpu::Expr) Nat),
    machine : Engine.Gpu::Machine,
    stmt : Engine.Gpu::Stmt
  ],
  result: Engine.Gpu::Machine,
  body: (match fuel
    | Nat.zero => machine
    | Nat.succ rest => (match stmt
      | Engine.Gpu::Stmt.skip => machine
      | Engine.Gpu::Stmt.assign slot expr => (Engine.Gpu::withLocal machine slot (apply ev machine expr))
      | Engine.Gpu::Stmt.store buffer index expr => (Engine.Gpu::withStore machine buffer (apply ev machine index) (apply ev machine expr))
      | Engine.Gpu::Stmt.seq before after => (exec rest ev (exec rest ev machine before) after)
      | Engine.Gpu::Stmt.when condition body => (if ((apply ev machine condition) =? 0) then machine else (exec rest ev machine body))
      | Engine.Gpu::Stmt.repeat start count slot body => (match count
        | Nat.zero => machine
        | Nat.succ remaining => (exec rest ev (exec rest ev (Engine.Gpu::withLocal machine slot start) body) (ctor Engine.Gpu::Stmt.repeat (start + 1) remaining slot body))))),
  recursive_argument: fuel
}

definition{
  name: atInvocation,
  parameters: [machine : Engine.Gpu::Machine, fresh : (List Nat), index : Nat],
  result: Engine.Gpu::Machine,
  body: (record Engine.Gpu::Machine {buffers := (project machine buffers), locals := fresh, invocation := index, group := 0})
}

definition{
  name: runAll,
  parameters: [
    fuel : Nat,
    ev : (Fn (Engine.Gpu::Machine Engine.Gpu::Expr) Nat),
    count : Nat,
    machine : Engine.Gpu::Machine,
    stmt : Engine.Gpu::Stmt
  ],
  result: Engine.Gpu::Machine,
  body: (match count
    | Nat.zero => machine
    | Nat.succ earlier => (exec fuel ev (atInvocation (runAll fuel ev earlier machine stmt) (project machine locals) earlier) stmt)),
  recursive_argument: count
}
\end{semanticmodule}
\end{lexlean}
