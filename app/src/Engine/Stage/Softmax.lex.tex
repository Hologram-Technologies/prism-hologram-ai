\title{Natural numbers}
\begin{semanticmodule}
definition{
  name: keyAt,
  parameters: [words : (List Nat), i : Nat],
  result: Nat,
  body: (prim remainder (((prim remainder (Engine.Stage.Greedy::nth words (prim multiply 2 i : Nat)) 4294967296 0 : Nat) + (prim multiply 4294967296 (prim remainder (Engine.Stage.Greedy::nth words ((prim multiply 2 i : Nat) + 1)) 4294967296 0 : Nat) : Nat)) + 9223372036854775808) 18446744073709551616 0 : Nat)
}

definition{
  name: topOf,
  parameters: [words : (List Nat), count : Nat],
  result: Nat,
  body: (match count
    | Nat.zero => 0
    | Nat.succ rest => (Engine.Stage.Greedy::maxOf (topOf words rest) (keyAt words rest))),
  recursive_argument: count
}

definition{
  name: weightOf,
  parameters: [distance : Nat],
  result: Nat,
  body: ((Engine.Stage.Exp2::exp2Neg distance) + (prim subtract (prim subtract 1073741824 (Engine.Stage.Exp2::exp2Neg distance) : Nat) (prim multiply 1073741824 (prim subtract distance (prim subtract distance 1 : Nat) : Nat) : Nat) : Nat))
}

definition{
  name: weightAt,
  parameters: [words : (List Nat), top : Nat, i : Nat],
  result: Nat,
  body: (weightOf (prim subtract top (keyAt words i) : Nat))
}

definition{
  name: totalOf,
  parameters: [words : (List Nat), top : Nat, count : Nat],
  result: Nat,
  body: (match count
    | Nat.zero => 0
    | Nat.succ rest => ((weightAt words top rest) + (totalOf words top rest))),
  recursive_argument: count
}

definition{
  name: divideEven,
  parameters: [n : Nat, d : Nat],
  result: Nat,
  body: ((prim quotient n d 0 : Nat) + (Engine.Stage.Exp2::roundUp (prim quotient n d 0 : Nat) (prim multiply 2 (prim remainder n d 0 : Nat) : Nat) d))
}

definition{
  name: share,
  parameters: [words : (List Nat), top : Nat, total : Nat, i : Nat],
  result: Nat,
  body: (divideEven (prim multiply 1073741824 (weightAt words top i) : Nat) total)
}

definition{
  name: runFrom,
  parameters: [words : (List Nat), top : Nat, total : Nat, i : Nat, remaining : Nat],
  result: (List Nat),
  body: (match remaining
    | Nat.zero => (nil Nat)
    | Nat.succ rest => (cons (share words top total i) (runFrom words top total (i + 1) rest))),
  recursive_argument: remaining
}

definition{
  name: runAt,
  parameters: [words : (List Nat), count : Nat, top : Nat],
  result: (List Nat),
  body: (runFrom words top (totalOf words top count) 0 count)
}

definition{
  name: run,
  parameters: [words : (List Nat), count : Nat],
  result: (List Nat),
  body: (runAt words count (topOf words count))
}
\end{semanticmodule}
