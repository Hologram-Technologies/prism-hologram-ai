\title{Natural numbers}
\begin{semanticmodule}
definition{
  name: valueAt,
  parameters: [words : (List Nat), i : Nat],
  result: Nat,
  body: ((prim remainder (Engine.Stage.Greedy::nth words (prim multiply 2 i : Nat)) 4294967296 0 : Nat) + (prim multiply 4294967296 (prim remainder (Engine.Stage.Greedy::nth words ((prim multiply 2 i : Nat) + 1)) 4294967296 0 : Nat) : Nat))
}

definition{
  name: negative,
  parameters: [v : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::below 9223372036854775807 v)
}

definition{
  name: size,
  parameters: [v : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::pick (negative v) (prim subtract 18446744073709551616 v : Nat) v)
}

definition{
  name: squaresFrom,
  parameters: [values : (List Nat), i : Nat, remaining : Nat],
  result: Nat,
  body: (match remaining
    | Nat.zero => 0
    | Nat.succ rest => ((prim multiply (size (valueAt values i)) (size (valueAt values i)) : Nat) + (squaresFrom values (i + 1) rest))),
  recursive_argument: remaining
}

definition{
  name: rootStep,
  parameters: [rest : Nat, root : Nat, bit : Nat, steps : Nat],
  result: Nat,
  body: (match steps
    | Nat.zero => root
    | Nat.succ more => (Engine.Stage.Greedy::pick (Engine.Stage.Greedy::below rest (root + bit)) (rootStep rest (prim quotient root 2 0 : Nat) (prim quotient bit 4 0 : Nat) more) (rootStep (prim subtract rest (root + bit) : Nat) ((prim quotient root 2 0 : Nat) + bit) (prim quotient bit 4 0 : Nat) more))),
  recursive_argument: steps
}

definition{
  name: integerRoot,
  parameters: [v : Nat],
  result: Nat,
  body: (rootStep v 0 85070591730234615865843651857942052864 64)
}

definition{
  name: meanRoot,
  parameters: [values : (List Nat), count : Nat, epsilon : Nat],
  result: Nat,
  body: (integerRoot (prim quotient (prim multiply ((squaresFrom values 0 count) + (prim multiply count epsilon : Nat)) 4294967296 : Nat) count 0 : Nat))
}

definition{
  name: scaled,
  parameters: [x : Nat, g : Nat, r : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::pick (prim remainder ((negative x) + (negative g)) 2 0 : Nat) (prim remainder (prim subtract 18446744073709551616 (Engine.Stage.Softmax::divideEven (prim multiply (prim multiply (size x) (size g) : Nat) 65536 : Nat) r) : Nat) 18446744073709551616 0 : Nat) (Engine.Stage.Softmax::divideEven (prim multiply (prim multiply (size x) (size g) : Nat) 65536 : Nat) r))

definition{
  name: outFrom,
  parameters: [values : (List Nat), weights : (List Nat), r : Nat, i : Nat, remaining : Nat],
  result: (List Nat),
  body: (match remaining
    | Nat.zero => (nil Nat)
    | Nat.succ rest => (cons (prim remainder (scaled (valueAt values i) (valueAt weights i) r) 4294967296 0 : Nat) (cons (prim quotient (scaled (valueAt values i) (valueAt weights i) r) 4294967296 0 : Nat) (outFrom values weights r (i + 1) rest)))),
  recursive_argument: remaining
}

definition{
  name: run,
  parameters: [values : (List Nat), weights : (List Nat), count : Nat, epsilon : Nat],
  result: (List Nat),
  body: (match (meanRoot values count epsilon)
    | Nat.zero => (nil Nat)
    | Nat.succ less => (outFrom values weights (meanRoot values count epsilon) 0 count))
}
