definition{
  name: eval,
  parameters: [machine : Machine, expr : Expr],
  result: Nat,
  body: (match expr
    | Expr.lit value => (word value)
    | Expr.invocation => (project machine invocation)
    | Expr.group => (project machine group)
    | Expr.local slot => (nth (project machine locals) slot)
    | Expr.load buffer index => (nth (nthBuffer (project machine buffers) buffer) (eval machine index))
    | Expr.add left right => (word ((eval machine left) + (eval machine right)))
    | Expr.sub left right => (word (prim subtract ((eval machine left) + 4294967296) (eval machine right) : Nat))
    | Expr.mul left right => (word (prim multiply (eval machine left) (eval machine right) : Nat))
    | Expr.div left right => (u32div (eval machine left) (eval machine right))
    | Expr.rem left right => (u32rem (eval machine left) (eval machine right))
    | Expr.shr left right => (u32shr (eval machine left) (eval machine right))
    | Expr.shl left right => (u32shl (eval machine left) (eval machine right))
    | Expr.band left right => (u32and (eval machine left) (eval machine right))
    | Expr.less left right => (if ((eval machine left) <? (eval machine right)) then 1 else 0)
    | Expr.select condition yes no => (if ((eval machine condition) =? 0) then (eval machine no) else (eval machine yes))),
  recursive_argument: expr
}

definition{
  name: wgslEval,
  parameters: [machine : Machine, term : Wgsl],
  result: Nat,
  body: (match term
    | Wgsl.wLit value => value
    | Wgsl.wLid => (project machine invocation)
    | Wgsl.wWid => (project machine group)
    | Wgsl.wLocal slot => (nth (project machine locals) slot)
    | Wgsl.wIndex buffer index => (nth (nthBuffer (project machine buffers) buffer) (wgslEval machine index))
    | Wgsl.wAdd left right => (prim remainder ((wgslEval machine left) + (wgslEval machine right)) 4294967296 0 : Nat)
    | Wgsl.wSub left right => (prim remainder (prim subtract ((wgslEval machine left) + 4294967296) (wgslEval machine right) : Nat) 4294967296 0 : Nat)
    | Wgsl.wMul left right => (prim remainder (prim multiply (wgslEval machine left) (wgslEval machine right) : Nat) 4294967296 0 : Nat)
    | Wgsl.wDiv left right => (u32div (wgslEval machine left) (wgslEval machine right))
    | Wgsl.wRem left right => (u32rem (wgslEval machine left) (wgslEval machine right))
    | Wgsl.wShr left right => (u32shr (wgslEval machine left) (wgslEval machine right))
    | Wgsl.wShl left right => (u32shl (wgslEval machine left) (wgslEval machine right))
    | Wgsl.wAnd left right => (u32and (wgslEval machine left) (wgslEval machine right))
    | Wgsl.wSelectLt otherwise chosen left right => (if ((wgslEval machine left) <? (wgslEval machine right)) then (wgslEval machine chosen) else (wgslEval machine otherwise))

definition{
  name: toWgsl,
  parameters: [expr : Expr],
  result: Wgsl,
  body: (match expr
    | Expr.lit value => (ctor Wgsl.wLit (word value))
    | Expr.invocation => (ctor Wgsl.wLid)

    | Expr.sub left right => (ctor Wgsl.wSub (toWgsl left) (toWgsl right))
    | Expr.mul left right => (ctor Wgsl.wMul (toWgsl left) (toWgsl right))
    | Expr.div left right => (ctor Wgsl.wDiv (toWgsl left) (toWgsl right))

    | Expr.band left right => (ctor Wgsl.wAnd (toWgsl left) (toWgsl right))

definition{
  name: exec,
  parameters: [fuel : Nat, ev : (Fn (Machine Expr) Nat), machine : Machine, stmt : Stmt],
  result: Machine,
  body: (match fuel
    | Nat.zero => machine
    | Nat.succ rest => (match stmt
      | Stmt.skip => machine
      | Stmt.assign slot expr => (withLocal machine slot (apply ev machine expr))
      | Stmt.store buffer index expr => (withStore machine buffer (apply ev machine index) (apply ev machine expr))
      | Stmt.seq before after => (exec rest ev (exec rest ev machine before) after)
      | Stmt.when condition body => (if ((apply ev machine condition) =? 0) then machine else (exec rest ev machine body))
      | Stmt.repeat start count slot body => (match count
        | Nat.zero => machine
        | Nat.succ remaining => (exec rest ev (exec rest ev (withLocal machine slot start) body) (ctor Stmt.repeat (start + 1) remaining slot body))))),
  recursive_argument: fuel
}

}

  name: foldE,
  parameters: [expr : Expr],
  result: Nat,
  body: (eval (record Machine {buffers := (nil (List Nat)), locals := (nil Nat), invocation := 0, group := 0}) expr)
}

definition{
  name: renderE,
  parameters: [slots : Nat, expr : Expr],
  result: Bytes,
  body: (match expr
    | Expr.lit value => (wlit value)
    | Expr.invocation => (text "k")
    | Expr.group => (text "0u")
    | Expr.local slot => (if (slot <? slots) then (cat (text "l[") (cat (decimal slot) (text "]"))) else (text "0u"))
    | Expr.load buffer index => (cat (text "rd") (cat (decimal buffer) (cat (text "(") (cat (renderE slots index) (text ")")))))
    | Expr.add left right => (if (closedE expr) then (wlit (foldE expr)) else (cat (text "(") (cat (renderE slots left) (cat (text " + ") (cat (renderE slots right) (text ")"))))))
    | Expr.sub left right => (if (closedE expr) then (wlit (foldE expr)) else (cat (text "(") (cat (renderE slots left) (cat (text " - ") (cat (renderE slots right) (text ")"))))))
    | Expr.mul left right => (if (closedE expr) then (wlit (foldE expr)) else (cat (text "(") (cat (renderE slots left) (cat (text " * ") (cat (renderE slots right) (text ")"))))))
    | Expr.div left right => (if (closedE expr) then (wlit (foldE expr)) else (if (closedE right) then (if ((foldE right) =? 0) then (renderE slots left) else (cat (text "(") (cat (renderE slots left) (cat (text " / ") (cat (renderE slots right) (text ")")))))) else (cat (text "(") (cat (renderE slots left) (cat (text " / ") (cat (renderE slots right) (text ")")))))))
    | Expr.rem left right => (if (closedE expr) then (wlit (foldE expr)) else (if (closedE right) then (if ((foldE right) =? 0) then (text "0u") else (cat (text "(") (cat (renderE slots left) (cat 0x202520 (cat (renderE slots right) (text ")")))))) else (cat (text "(") (cat (renderE slots left) (cat 0x202520 (cat (renderE slots right) (text ")")))))))
    | Expr.shr left right => (if (closedE expr) then (wlit (foldE expr)) else (cat (text "(") (cat (renderE slots left) (cat (text " >> (") (cat (renderE slots right) (text " & 31u))"))))))
    | Expr.shl left right => (if (closedE expr) then (wlit (foldE expr)) else (cat (text "(") (cat (renderE slots left) (cat (text " << (") (cat (renderE slots right) (text " & 31u))"))))))
    | Expr.band left right => (if (closedE expr) then (wlit (foldE expr)) else (cat (text "(") (cat (renderE slots left) (cat (text " & ") (cat (renderE slots right) (text ")"))))))
    | Expr.less left right => (if (closedE expr) then (wlit (foldE expr)) else (cat (text "select(0u, 1u, ") (cat (renderE slots left) (cat (text " < ") (cat (renderE slots right) (text ")"))))))
    | Expr.select condition yes no => (if (closedE condition) then (if ((foldE condition) =? 0) then (renderE slots no) else (renderE slots yes)) else (cat (text "select(") (cat (renderE slots no) (cat (text ", ") (cat (renderE slots yes) (cat (text ", ") (cat (renderE slots condition) (text " != 0u)"))))))))),
  recursive_argument: expr
}

definition{
  name: renderS,
  parameters: [slots : Nat, depth : Nat, stmt : Stmt],
  result: Bytes,
  body: (match stmt
    | Stmt.skip => (text "")
    | Stmt.assign slot expr => (if (slot <? slots) then (cat (text "l[") (cat (decimal slot) (cat (text "] = ") (cat (renderE slots expr) (text ";\n"))))) else (text ""))
    | Stmt.store buffer index expr => (cat (text "{ let at = ") (cat (renderE slots index) (cat (text "; let put = ") (cat (renderE slots expr) (cat (text "; if (at < arrayLength(&b") (cat (decimal buffer) (cat (text ")) { b") (cat (decimal buffer) (text "[at] = put; } }\n")))))))))
    | Stmt.seq before after => (cat (renderS slots depth before) (renderS slots depth after))
    | Stmt.when condition body => (cat (text "if (") (cat (renderE slots condition) (cat (text " != 0u) {\n") (cat (renderS slots depth body) (text "}\n")))))
    | Stmt.repeat start count slot body => (cat (text "for (var ") (cat (cat (text "i") (decimal depth)) (cat (text ": u32 = 0u; ") (cat (cat (text "i") (decimal depth)) (cat (text " < ") (cat (wlit count) (cat (text "; ") (cat (cat (text "i") (decimal depth)) (cat (text " = ") (cat (cat (text "i") (decimal depth)) (cat (text " + 1u) {\n") (cat (if (slot <? slots) then (cat (text "l[") (cat (decimal slot) (cat (text "] = ") (cat (wlit start) (cat (text " + ") (cat (cat (text "i") (decimal depth)) (text ";\n"))))))) else (text "")) (cat (renderS slots (depth + 1) body) (text "}\n"))))))))))))))),
  recursive_argument: stmt
}

definition{
  name: writes,
  parameters: [stmt : Stmt, buffer : Nat],
  result: Bool,
  body: (match stmt
    | Stmt.skip => false
    | Stmt.assign slot expr => false
    | Stmt.store target index expr => (target =? buffer)
    | Stmt.seq before after => (if (writes before buffer) then true else (writes after buffer))
    | Stmt.when condition body => (writes body buffer)
    | Stmt.repeat start count slot body => (writes body buffer)),
  recursive_argument: stmt
}

definition{
  name: bindings,
  parameters: [buffer : Nat, left : Nat, stmt : Stmt],
  result: Bytes,
  body: (match left
    | Nat.zero => (text "")
    | Nat.succ rest => (cat (cat (text "@group(0) @binding(") (cat (decimal buffer) (cat (text ") var<storage, ") (cat (if (writes stmt buffer) then (text "read_write") else (text "read")) (cat (text "> b") (cat (decimal buffer) (cat (text ": array<u32>;\nfn rd") (cat (decimal buffer) (cat (text "(i: u32) -> u32 { if (i < arrayLength(&b") (cat (decimal buffer) (cat (text ")) { return b") (cat (decimal buffer) (text "[i]; } return 0u; }\n"))))))))))))) (bindings (buffer + 1) rest stmt))),
  recursive_argument: left
}

definition{
  name: renderKernel,
  parameters: [buffers : Nat, slots : Nat, stmt : Stmt],
  result: Bytes,
  body: (cat (bindings 0 buffers stmt) (cat (text "override invocations: u32;\n@compute @workgroup_size(64)\nfn main(@builtin(global_invocation_id) gid: vec3<u32>) {\nlet k = gid.x;\nif (k >= invocations) { return; }\nvar l: array<u32, ") (cat (decimal (if (slots =? 0) then 1 else slots)) (cat (text ">;\n") (cat (renderS slots 0 stmt) (text "}\n"))))))
}
