definition{
  name: shiftRight,
  parameters: [value : UInt32, amount : UInt32],
  result: UInt32,
  body: (match (prim shift_right value amount : (Option UInt32))
    | Option.some shifted => shifted
    | Option.none => (0 : UInt32)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: shiftLeft,
  parameters: [value : UInt32, amount : UInt32],
  result: UInt32,

    | Option.none => (0 : UInt32)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: half,
