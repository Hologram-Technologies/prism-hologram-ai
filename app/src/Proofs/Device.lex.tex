theorem{
  name: device_webgpu_decides,
  parameters: [present : Nat, limit : Nat, largest : Nat, model : Nat, ceiling : Nat],
  statement: (((Kernel.Device::deviceCode present limit largest model ceiling) = 2) <-> ((Kernel.Backend::gpuUsable present limit largest) = true)),
  proof: linear_arithmetic{definitions: [Kernel.Backend::gpuUsable, Kernel.Device::deviceCode]},
  axioms: [Classical.choice, Quot.sound, propext]

theorem{
  name: device_cpu_decides,
  parameters: [present : Nat, limit : Nat, largest : Nat, model : Nat, ceiling : Nat],
  statement: (((Kernel.Device::deviceCode present limit largest model ceiling) = 1) <-> (((Kernel.Backend::gpuUsable present limit largest) = false) /\ (model <= ceiling))),
  proof: linear_arithmetic{definitions: [Kernel.Backend::gpuUsable, Kernel.Device::deviceCode]},
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
