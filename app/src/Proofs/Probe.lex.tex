\title{Bounded byte buffer}
\begin{semanticmodule}
theorem{
  name: probe_exact_decides,
  parameters: [
    layer : Nat,
    layers : Nat,
    position : Nat,
    positions : Nat,
    component : Nat,
    components : Nat,
    classOfRun : Nat,
    sameAdapter : Nat
  ],
  statement: (((Kernel.Probe::probeCode layer layers position positions component components classOfRun sameAdapter) = 2) <-> (((layer + 1) <= layers) /\ (((position + 1) <= positions) /\ (((component + 1) <= components) /\ ((Kernel.Determinism::comparable classOfRun sameAdapter) = true))))),
  proof: linear_arithmetic{
    definitions: [
      Kernel.Determinism::comparable,
      Kernel.Probe::probeCode,
      Kernel.Probe::probeInRange
    ]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}
