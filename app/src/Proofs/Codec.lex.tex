\title{Bounded byte buffer}
\begin{semanticmodule}
theorem{
  name: codec_round_trip,
  parameters: [format : Nat, stored : Nat],
  statement: ((format <= 3) -> ((stored < (Kernel.Codec::codecWidth format)) -> ((Kernel.Codec::codecEncode format (Kernel.Codec::codecHigh format stored) (Kernel.Codec::codecMiddle format stored) (Kernel.Codec::codecLow format stored)) = stored))),
  proof: linear_arithmetic{
    definitions: [
      Kernel.Codec::codecEncode,
      Kernel.Codec::codecHigh,
      Kernel.Codec::codecLow,
      Kernel.Codec::codecLowSize,
      Kernel.Codec::codecMiddle,
      Kernel.Codec::codecMiddleSize,
      Kernel.Codec::codecWidth
    ]
  },
  axioms: [Quot.sound, propext]
}

theorem{
  name: codec_fields_fit,
  parameters: [format : Nat, stored : Nat],
  statement: (((Kernel.Codec::codecLow format stored) < (Kernel.Codec::codecLowSize format)) /\ ((Kernel.Codec::codecMiddle format stored) < (Kernel.Codec::codecMiddleSize format))),
  proof: linear_arithmetic{
    definitions: [
      Kernel.Codec::codecLow,
      Kernel.Codec::codecLowSize,
      Kernel.Codec::codecMiddle,
      Kernel.Codec::codecMiddleSize
    ]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}
