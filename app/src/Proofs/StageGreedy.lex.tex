theorem{
  name: key_of_pick,
  parameters: [logits : (List Nat), choice : Bool, chosen : Nat, other : Nat],
  statement: ((Engine.Stage.Greedy::keyAt logits (Engine.Stage.Greedy::pick choice chosen other)) = (Engine.Stage.Greedy::pick choice (Engine.Stage.Greedy::keyAt logits chosen) (Engine.Stage.Greedy::keyAt logits other))),
  proof: cases{
    branches: [
      {
        proof: simplify{definitions: [Engine.Stage.Greedy::pick]},
        binders: [],
        constructor: "false"
      },
      {proof: simplify{definitions: [Engine.Stage.Greedy::pick]}, binders: [], constructor: "true"}
    ],
    scrutinee: choice
  },
  axioms: [propext]
}

theorem{
  name: pick_is_max,
  parameters: [choice : Bool, kept : Nat, next : Nat],
  statement: (((kept <? next) = choice) -> ((Engine.Stage.Greedy::pick choice next kept) = (Engine.Stage.Greedy::maxOf kept next))),
  proof: cases{
    branches: [
      {
        proof: linear_arithmetic{
          definitions: [Engine.Stage.Greedy::maxOf, Engine.Stage.Greedy::pick]
        },
        binders: [],
        constructor: "false"
      },
      {
        proof: linear_arithmetic{
          definitions: [Engine.Stage.Greedy::maxOf, Engine.Stage.Greedy::pick]
        },
        binders: [],
        constructor: "true"
      }
    ],
    scrutinee: choice
  },
  axioms: [Quot.sound, propext]
}
