definition{
  name: four,
  parameters: [],
  result: (List Nat),
  body: (cons 0 (cons 0 (cons 0 (cons 0 (nil Nat)))))
}

definition{
  name: cap,
  parameters: [count : Nat],
  result: Nat,
  body: (match count | Nat.zero => 0 | Nat.succ rest => (524288 + (cap rest))),
  recursive_argument: count
}

definition{
  name: times,
  parameters: [a : Nat, w : Nat],
  result: Nat,
  body: (Engine.Word::mulB (four) a w)
}

definition{
  name: actNeg,
  parameters: [code : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::below 32767 (prim remainder code 65536 0 : Nat))
}

definition{
  name: actMag,
  parameters: [code : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::pick (actNeg code) (prim subtract 65536 (prim remainder code 65536 0 : Nat) : Nat) (prim remainder code 65536 0 : Nat))
}

definition{
  name: shiftDown,
  parameters: [value : Nat, count : Nat],
  result: Nat,
  body: (match count
    | Nat.zero => value
    | Nat.succ rest => (shiftDown (prim quotient value 16 0 : Nat) rest)),
  recursive_argument: count
}

definition{
  name: nibble,
  parameters: [weights : (List Nat), g : Nat],
  result: Nat,
  body: (prim remainder (shiftDown (Engine.Stage.Greedy::nth weights (prim quotient g 8 0 : Nat)) (prim remainder g 8 0 : Nat)) 16 0 : Nat)
}

definition{
  name: weightNeg,
  parameters: [n : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::below n 8)
}

definition{
  name: weightMag,
  parameters: [n : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::pick (weightNeg n) (prim subtract 8 n : Nat) (prim subtract n 8 : Nat))
}

definition{
  name: actAt,
  parameters: [acts : (List Nat), units : Nat, k : Nat, i : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::nth acts ((prim multiply 32 (prim remainder k units 0 : Nat) : Nat) + i))
}

definition{
  name: nibbleAt,
  parameters: [weights : (List Nat), k : Nat, i : Nat],
  result: Nat,
  body: (nibble weights ((prim multiply 32 k : Nat) + i))
}

definition{
  name: termNeg,
  parameters: [code : Nat, n : Nat],
  result: Nat,
  body: (prim remainder ((actNeg code) + (weightNeg n)) 2 0 : Nat)
}

definition{
  name: termMag,
  parameters: [code : Nat, n : Nat],
  result: Nat,
  body: (times (actMag code) (weightMag n))
}

definition{
  name: posPart,
  parameters: [code : Nat, n : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::pick (termNeg code n) 0 (termMag code n))
}

definition{
  name: negPart,
  parameters: [code : Nat, n : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::pick (termNeg code n) (termMag code n) 0)
}

definition{
  name: posSum,
  parameters: [acts : (List Nat), weights : (List Nat), units : Nat, k : Nat, count : Nat],
  result: Nat,
  body: (match count
    | Nat.zero => 0
    | Nat.succ rest => ((posPart (actAt acts units k rest) (nibbleAt weights k rest)) + (posSum acts weights units k rest))),
  recursive_argument: count
}

definition{
  name: negSum,
  parameters: [acts : (List Nat), weights : (List Nat), units : Nat, k : Nat, count : Nat],
  result: Nat,
  body: (match count
    | Nat.zero => 0
    | Nat.succ rest => ((negPart (actAt acts units k rest) (nibbleAt weights k rest)) + (negSum acts weights units k rest))),
  recursive_argument: count
}

definition{
  name: encode,
  parameters: [p : Nat, n : Nat],
  result: Nat,
  body: (Engine.Stage.Greedy::pick (Engine.Stage.Greedy::below p n) (prim subtract 4294967296 (prim subtract n p : Nat) : Nat) (prim subtract p n : Nat))
}

definition{
  name: groupSum,
  parameters: [acts : (List Nat), weights : (List Nat), units : Nat, k : Nat],
  result: Nat,
  body: (encode (posSum acts weights units k 32) (negSum acts weights units k 32))
}

definition{
  name: runFrom,
  parameters: [acts : (List Nat), weights : (List Nat), units : Nat, k : Nat, remaining : Nat],
  result: (List Nat),
  body: (match remaining
    | Nat.zero => (nil Nat)
    | Nat.succ rest => (cons (groupSum acts weights units k) (runFrom acts weights units (k + 1) rest))),
  recursive_argument: remaining
}

definition{
  name: run,
  parameters: [acts : (List Nat), weights : (List Nat), units : Nat, count : Nat],
  result: (List Nat),
  body: (runFrom acts weights units 0 count)
}
