definition{
  name: powBy,
  parameters: [base : Nat, exponent : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => 1
    | Nat.succ rest => (prim multiply (Engine.Stage.Greedy::pick (prim remainder exponent 2 0 : Nat) base 1) (powBy (prim multiply base base : Nat) (prim quotient exponent 2 0 : Nat) rest) : Nat)),
  recursive_argument: fuel
}

definition{
  name: certified,
  parameters: [fraction : Nat],
  result: Bool,
  body: (((powBy (prim subtract (prim multiply 2 (Engine.Stage.Exp2::entry fraction) : Nat) 1 : Nat) 256 9) <? (powBy 2 (prim subtract 7936 fraction : Nat) 13)) && ((powBy 2 (prim subtract 7936 fraction : Nat) 13) <? (powBy ((prim multiply 2 (Engine.Stage.Exp2::entry fraction) : Nat) + 1) 256 9)))
}

definition{
  name: certifiedBelow,
  parameters: [count : Nat],
  result: Bool,
  body: (match count
    | Nat.zero => true
    | Nat.succ rest => ((certified rest) && (certifiedBelow rest))),
  recursive_argument: count
}

definition{
  name: distanceTwice,
  parameters: [value : Nat, shift : Nat, out : Nat],
  result: Nat,
  body: (prim multiply 2 ((prim subtract value (prim multiply out (Engine.Stage.Exp2::pow2 shift) : Nat) : Nat) + (prim subtract (prim multiply out (Engine.Stage.Exp2::pow2 shift) : Nat) value : Nat)) : Nat)
}

definition{
  name: rounded,
  parameters: [value : Nat, shift : Nat, out : Nat],
  result: Bool,
  body: (((distanceTwice value shift out) <=? (Engine.Stage.Exp2::pow2 shift)) && ((not ((distanceTwice value shift out) =? (Engine.Stage.Exp2::pow2 shift))) || ((prim remainder out 2 0 : Nat) =? 0)))
}

definition{
  name: roundedBelow,
  parameters: [shift : Nat, count : Nat],
  result: Bool,
  body: (match count
    | Nat.zero => true
    | Nat.succ rest => ((rounded (Engine.Stage.Exp2::entry rest) shift (Engine.Stage.Exp2::shifted (Engine.Stage.Exp2::entry rest) shift)) && (roundedBelow shift rest))),
  recursive_argument: count
}
