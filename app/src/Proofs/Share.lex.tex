\title{Bounded byte buffer}
\begin{semanticmodule}
theorem{
  name: shareable_decides,
  parameters: [
    asked : Kernel.Admission::Label,
    held : Kernel.Admission::Label,
    heldLength : Nat,
    promptLength : Nat,
    origin : Nat,
    classOfRun : Nat,
    sameAdapter : Nat
  ],
  statement: (((Kernel.Share::shareable asked held heldLength promptLength origin classOfRun sameAdapter) = true) <-> (((Kernel.Reuse::reusable asked held heldLength promptLength) = true) /\ (((origin = 0) -> (1 = 0)) -> ((Kernel.Determinism::comparable classOfRun sameAdapter) = true)))),
  proof: linear_arithmetic{
    definitions: [
      Kernel.Admission::sameLabel,
      Kernel.Determinism::comparable,
      Kernel.Reuse::reusable,
      Kernel.Reuse::reuseShorter,
      Kernel.Share::shareable
    ]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}
\end{semanticmodule}
