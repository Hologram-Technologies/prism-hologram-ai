\begin{lexlean}{Proofs.Opnames}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Kernel.Opnames}
\title{Bounded byte buffer}
\begin{semanticmodule}
theorem{
  name: opname_is_decides,
  parameters: [wa : Nat, wb : Nat, wc : Nat, wd : Nat, a : Nat, b : Nat, c : Nat, d : Nat],
  statement: (((Kernel.Opnames::opnameIs wa wb wc wd a b c d) = true) <-> ((wa = a) /\ ((wb = b) /\ ((wc = c) /\ (wd = d))))),
  proof: linear_arithmetic{definitions: [Kernel.Opnames::opnameIs]},
  axioms: [propext]
}

theorem{
  name: opname_listed_encoder,
  parameters: [wa : Nat, wb : Nat, wc : Nat, wd : Nat],
  statement: ((Kernel.Opnames::opnameListed 2 wa wb wc wd) = (Kernel.Opnames::opnameEncoder wa wb wc wd)),
  proof: simplify{definitions: [Kernel.Opnames::opnameListed]},
  axioms: [propext]
}

theorem{
  name: opname_listed_unknown,
  parameters: [style : Nat, wa : Nat, wb : Nat, wc : Nat, wd : Nat],
  statement: ((3 <= style) -> ((Kernel.Opnames::opnameListed style wa wb wc wd) = false)),
  proof: linear_arithmetic{definitions: [Kernel.Opnames::opnameListed]},
  axioms: [Classical.choice, Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
