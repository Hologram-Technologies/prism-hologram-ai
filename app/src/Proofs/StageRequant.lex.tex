definition{
  name: signed16,
  parameters: [v : Nat],
  result: Bool,
  body: ((v <=? 32767) || (4294934529 <=? v))
}

definition{
  name: allCodes,
  parameters: [codes : (List Nat)],
  result: Bool,
  body: (match codes
    | List.nil => true
    | List.cons head tail => ((signed16 head) && (allCodes tail))),
  recursive_argument: codes
}
