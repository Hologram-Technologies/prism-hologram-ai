theorem{
  name: stream_cut,
  parameters: [early : (List Nat), state : Engine.Stream::HashState, later : (List Nat)],
  statement: ((Engine.Stream::feed (Engine.Stream::feed state early) later) = (Engine.Stream::feed state (Engine.Stream::join early later))),
  proof: induction{
    branches: [
      {
        proof: simplify{definitions: [Engine.Stream::feed, Engine.Stream::join]},
        binders: [],
        constructor: nil
      },
      {
        proof: simplify{definitions: [hypothesis, Engine.Stream::feed, Engine.Stream::join]},
        binders: [head, rest, hypothesis],
        constructor: cons
      }
    ],
    generalizing: [state],
    scrutinee: early
  },
  axioms: [Quot.sound, propext]
}

theorem{
  name: stream_reads_its_chunk,
  parameters: [fuel : Nat, index : Nat, state : Engine.Stream::HashState, value : Bytes],
  statement: ((Kernel.Stream::feedFrom fuel state value index) = (Engine.Stream::feed state (Kernel.Stream::chunkOf fuel value index))),
  proof: induction{
    branches: [
      {
        proof: simplify{
          definitions: [Engine.Stream::feed, Kernel.Stream::chunkOf, Kernel.Stream::feedFrom]
        },
        binders: [],
        constructor: zero
      },
      {
        proof: simplify{
          definitions: [
            hypothesis,
            Engine.Stream::feed,
            Kernel.Stream::chunkOf,
            Kernel.Stream::feedFrom
          ]
        },
        binders: [rest, hypothesis],
        constructor: succ
      }
    ],
    generalizing: [index, state],
    scrutinee: fuel
  },
  axioms: [Quot.sound, propext]
}
