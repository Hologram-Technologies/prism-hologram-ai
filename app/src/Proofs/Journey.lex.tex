\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: readyBy,
  parameters: [state : Nat, event : Nat],
  result: Bool,
  body: (((state =? 4) && (event =? 5)) || ((state =? 6) && ((event =? 7) || (event =? 8))))
}

theorem{
  name: enters_preflight,
  parameters: [state : Nat, event : Nat],
  statement: (((Kernel.Journey::nextState state event) = 1) <-> ((state = 0) /\ (event = 1))),
  proof: linear_arithmetic{
    definitions: [Kernel.Journey::nextState, Kernel.Journey::working, readyBy]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}
