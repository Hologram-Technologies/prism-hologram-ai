theorem{
  name: integer_run_decides,
  parameters: [productsExact : Nat, restExact : Nat],
  statement: (((Kernel.Integer::integerRun productsExact restExact) = true) <-> ((productsExact = 1) /\ (restExact = 1))),
  proof: linear_arithmetic{
    definitions: [Kernel.Determinism::classCode, Kernel.Integer::integerRun]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: integer_run_is_everywhere,
  parameters: [productsExact : Nat, restExact : Nat, fixedOrder : Nat],
  statement: (((Kernel.Integer::integerRun productsExact restExact) = true) <-> ((Kernel.Determinism::classCode productsExact restExact fixedOrder) = 2)),
  proof: linear_arithmetic{
    definitions: [Kernel.Determinism::classCode, Kernel.Integer::integerRun]

  axioms: [Classical.choice, Quot.sound, propext]
