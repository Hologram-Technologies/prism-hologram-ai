\title{Bounded byte buffer}
\begin{semanticmodule}
theorem{
  name: nbits_shaped_decides,
  parameters: [part : Nat, rows : Nat, columns : Nat, size : Nat],
  statement: (((Kernel.Nbits::nbitsShaped part rows columns size) = true) <-> ((part <= 2) /\ ((1 <= rows) /\ ((rows <= 16777215) /\ ((1 <= columns) /\ ((columns <= 16777215) /\ ((Kernel.Nbits::nbitsBlock size) = true))))))),
  proof: linear_arithmetic{definitions: [Kernel.Nbits::nbitsBlock, Kernel.Nbits::nbitsShaped]},
  axioms: [propext]
}

theorem{
  name: nbits_block_bounds,
  parameters: [size : Nat],
  statement: (((Kernel.Nbits::nbitsBlock size) = true) -> ((16 <= size) /\ ((size <= 256) /\ ((prim remainder size 16 0 : Nat) = 0)))),
  proof: linear_arithmetic{definitions: [Kernel.Nbits::nbitsBlock]},
  axioms: [Classical.choice, Quot.sound, propext]
}

  name: nbits_blocks_fewest_16,
