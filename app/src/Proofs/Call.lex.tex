theorem{
  name: call_supported_runs,
  parameters: [task : Nat, device : Nat, dtype : Nat, beams : Nat],
  statement: ((task <= 2) -> ((device <= 4) -> ((dtype <= 2) -> ((beams <= 1) -> ((Kernel.Call::callCode task device dtype 0 beams 1) = (Kernel.Call::callDevice device)))))),
  proof: linear_arithmetic{definitions: [Kernel.Call::callCode, Kernel.Call::callDevice]},
  axioms: [Quot.sound, propext]
}
