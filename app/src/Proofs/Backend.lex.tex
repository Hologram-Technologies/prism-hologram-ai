\begin{semanticmodule}
theorem{
  name: gpu_usable_decides,
  parameters: [present : Nat, limit : Nat, largest : Nat],
  statement: (((Kernel.Backend::gpuUsable present limit largest) = true) <-> ((present = 1) /\ (largest <= limit))),
  proof: linear_arithmetic{definitions: [Kernel.Backend::gpuUsable]},
  axioms: [propext]
}

theorem{
  name: limit_monotone,
  parameters: [present : Nat, limit : Nat, wider : Nat, largest : Nat],
  statement: ((limit <= wider) -> (((Kernel.Backend::gpuUsable present limit largest) = true) -> ((Kernel.Backend::gpuUsable present wider largest) = true))),
  proof: linear_arithmetic{definitions: [Kernel.Backend::gpuUsable]},
  axioms: [Classical.choice, Quot.sound, propext]
}
