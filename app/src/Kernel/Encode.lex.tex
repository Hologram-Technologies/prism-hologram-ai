\begin{lexlean}{Kernel.Encode}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Kernel.Wire}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: encodeCode,
  parameters: [task : Nat, pooling : Nat, normalize : Nat],
  result: Nat,
  body: (if (task =? 1) then (if (pooling =? 1) then (if (normalize =? 1) then 0 else 2) else 1) else (if (task =? 2) then (if (pooling =? 0) then (if (normalize =? 0) then 0 else 2) else 1) else 3))
}

definition{
  name: encodeText,
  parameters: [code : Nat],
  result: Bytes,
  body: (if (code =? 0) then (prim utf8_encode "run" : Bytes) else (if (code =? 1) then (prim utf8_encode "refused: pooling" : Bytes) else (if (code =? 2) then (prim utf8_encode "refused: normalize" : Bytes) else (prim utf8_encode "refused: unsupported task" : Bytes))))
}

definition{
  name: encodeAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 4) then (encodeText (encodeCode (Kernel.Wire::fieldAt value 1 1) (Kernel.Wire::fieldAt value 2 1) (Kernel.Wire::fieldAt value 3 1))) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
