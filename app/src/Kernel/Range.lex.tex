\begin{lexlean}{Kernel.Range}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Kernel.Wire}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: within,
  parameters: [fileLength : Nat, previousEnd : Nat, offset : Nat, length : Nat],
  result: Bool,
  body: ((previousEnd <=? offset) && ((offset <=? fileLength) && (length <=? (prim subtract fileLength offset : Nat))))
}

structure{
  name: Span,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: fileLength, type: Nat},
    {name: previousEnd, type: Nat},
    {name: offset, type: Nat},
    {name: length, type: Nat}
  ]
}

definition{
  name: accepted,
  parameters: [span : Span],
  result: Bool,
  body: (within (project span fileLength) (project span previousEnd) (project span offset) (project span length))
}

definition{
  name: spanAt,
  parameters: [value : Bytes],
  result: Span,
  body: (record Span {fileLength := (Kernel.Wire::fieldAt value 1 8), previousEnd := (Kernel.Wire::fieldAt value 9 8), offset := (Kernel.Wire::fieldAt value 17 8), length := (Kernel.Wire::fieldAt value 25 8)}),
  axioms: [Quot.sound, propext]
}

definition{
  name: rangeAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 33) then (if (accepted (spanAt value)) then (prim utf8_encode "accept" : Bytes) else (prim utf8_encode "reject" : Bytes)) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
