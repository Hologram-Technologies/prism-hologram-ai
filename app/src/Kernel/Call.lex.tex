\begin{lexlean}{Kernel.Call}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Kernel.Wire}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: callDevice,
  parameters: [device : Nat],
  result: Nat,
  body: (if (device =? 0) then 0 else (if (device <=? 2) then 1 else (if (device <=? 4) then 2 else 3)))
}

definition{
  name: callCode,
  parameters: [task : Nat, device : Nat, dtype : Nat, sample : Nat, beams : Nat, penalty : Nat],
  result: Nat,
  body: (if (3 <=? task) then 10 else (if ((callDevice device) =? 3) then 11 else (if (3 <=? dtype) then 12 else (if (1 <=? sample) then 13 else (if (2 <=? beams) then 14 else (if (penalty =? 1) then (callDevice device) else 15))))))
}

definition{
  name: callText,
  parameters: [code : Nat],
  result: Bytes,
  body: (if (code =? 0) then (prim utf8_encode "auto" : Bytes) else (if (code =? 1) then (prim utf8_encode "webgpu" : Bytes) else (if (code =? 2) then (prim utf8_encode "cpu" : Bytes) else (if (code =? 10) then (prim utf8_encode "refused: unsupported task" : Bytes) else (if (code =? 11) then (prim utf8_encode "refused: unsupported device" : Bytes) else (if (code =? 12) then (prim utf8_encode "refused: unsupported dtype" : Bytes) else (if (code =? 13) then (prim utf8_encode "refused: sampling" : Bytes) else (if (code =? 14) then (prim utf8_encode "refused: beam search" : Bytes) else (prim utf8_encode "refused: repetition penalty" : Bytes)))))))))
}

definition{
  name: callAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 7) then (callText (callCode (Kernel.Wire::fieldAt value 1 1) (Kernel.Wire::fieldAt value 2 1) (Kernel.Wire::fieldAt value 3 1) (Kernel.Wire::fieldAt value 4 1) (Kernel.Wire::fieldAt value 5 1) (Kernel.Wire::fieldAt value 6 1))) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
