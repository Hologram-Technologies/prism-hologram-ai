\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: planBounded,
  parameters: [layerBytes : Nat, layers : Nat, stageLayers : Nat, stageCount : Nat],
  result: Bool,
  body: (((layers <=? 65536) && (layerBytes <=? 1099511627776)) && ((stageLayers <=? 65536) && (stageCount <=? 65536)))
}

definition{
  name: canRun,
  parameters: [budget : Nat, contextBytes : Nat, layerBytes : Nat],
  result: Bool,
  body: ((contextBytes <=? budget) && (layerBytes <=? (prim subtract budget contextBytes : Nat)))
}

definition{
  name: stageFits,
  parameters: [budget : Nat, contextBytes : Nat, layerBytes : Nat, stageLayers : Nat],
  result: Bool,
  body: ((contextBytes <=? budget) && ((prim multiply stageLayers layerBytes : Nat) <=? (prim subtract budget contextBytes : Nat)))
}

definition{
  name: stageLargest,
  parameters: [budget : Nat, contextBytes : Nat, layerBytes : Nat, layers : Nat, stageLayers : Nat],
  result: Bool,
  body: ((stageLayers =? layers) || ((prim subtract budget contextBytes : Nat) <? (prim multiply (stageLayers + 1) layerBytes : Nat)))
}

definition{
  name: stagesCover,
  parameters: [layers : Nat, stageLayers : Nat, stageCount : Nat],
  result: Bool,
  body: (((1 <=? stageCount) && ((prim multiply (prim subtract stageCount 1 : Nat) stageLayers : Nat) <? layers)) && (layers <=? (prim multiply stageCount stageLayers : Nat)))
}

definition{
  name: planned,
  parameters: [
    budget : Nat,
    contextBytes : Nat,
    layerBytes : Nat,
    layers : Nat,
    stageLayers : Nat,
    stageCount : Nat
  ],
  result: Bool,
  body: (((1 <=? stageLayers) && (stageLayers <=? layers)) && ((stageFits budget contextBytes layerBytes stageLayers) && ((stageLargest budget contextBytes layerBytes layers stageLayers) && (stagesCover layers stageLayers stageCount))))
}
