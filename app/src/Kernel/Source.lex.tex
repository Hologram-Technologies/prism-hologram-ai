definition{
  name: sourceCode,
  parameters: [task : Nat, precision : Nat, declared : Nat, part : Nat],
  result: Nat,
  body: (if (task <=? 2) then (if (precision <=? 3) then (if (declared <=? 100) then (if (part <=? declared) then 4 else 3) else 2) else 1) else 0)
}

definition{
  name: sourceGraph,
  parameters: [precision : Nat],
  result: Bytes,
  body: (if (precision =? 0) then (prim utf8_encode "onnx/model_q4f16.onnx" : Bytes) else (if (precision =? 1) then (prim utf8_encode "onnx/model_q4.onnx" : Bytes) else (if (precision =? 2) then (prim utf8_encode "onnx/model_fp16.onnx" : Bytes) else (prim utf8_encode "onnx/model.onnx" : Bytes))))
}

  name: sourceName,
  parameters: [precision : Nat, part : Nat],
  result: Bytes,
  body: (if (part =? 0) then (sourceGraph precision) else (if (part =? 1) then (prim append (sourceGraph precision) (prim utf8_encode "_data" : Bytes) : Bytes) else (prim append (prim append (sourceGraph precision) (prim utf8_encode "_data_" : Bytes) : Bytes) (Kernel.Wire::decimal (prim subtract part 1 : Nat)) : Bytes)))
}

definition{
  name: sourceText,
  parameters: [code : Nat, precision : Nat, part : Nat],
  result: Bytes,
  body: (if (code =? 4) then (sourceName precision part) else (if (code =? 3) then (prim utf8_encode "undeclared file" : Bytes) else (if (code =? 2) then (prim utf8_encode "too many files" : Bytes) else (if (code =? 1) then (prim utf8_encode "unsupported precision" : Bytes) else (prim utf8_encode "unsupported task" : Bytes)))))
}

definition{
  name: sourceAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 5) then (sourceText (sourceCode (Kernel.Wire::fieldAt value 1 1) (Kernel.Wire::fieldAt value 2 1) (Kernel.Wire::fieldAt value 3 1) (Kernel.Wire::fieldAt value 4 1)) (Kernel.Wire::fieldAt value 2 1) (Kernel.Wire::fieldAt value 4 1)) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
