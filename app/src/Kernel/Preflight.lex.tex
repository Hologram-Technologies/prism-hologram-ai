\begin{lexlean}{Kernel.Preflight}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Kernel.Wire}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: verdictCode,
  parameters: [recognised : Nat, missing : Nat, manifestFits : Nat, unsupported : Nat],
  result: Nat,
  body: (if ((recognised <=? 1) && (manifestFits <=? 1)) then (if (missing =? 0) then (if (unsupported =? 0) then (if (manifestFits =? 1) then (if (recognised =? 1) then 4 else 3) else 2) else 1) else 0) else 5)
}

definition{
  name: authorised,
  parameters: [recognised : Nat, missing : Nat, manifestFits : Nat, unsupported : Nat],
  result: Bool,
  body: (((verdictCode recognised missing manifestFits unsupported) =? 3) || ((verdictCode recognised missing manifestFits unsupported) =? 4))
}

definition{
  name: verdictText,
  parameters: [code : Nat],
  result: Bytes,
  body: (if (code =? 4) then (prim utf8_encode "supported" : Bytes) else (if (code =? 3) then (prim utf8_encode "derived" : Bytes) else (if (code =? 2) then (prim utf8_encode "rejected: manifest does not fit" : Bytes) else (if (code =? 1) then (prim utf8_encode "rejected: unsupported feature" : Bytes) else (if (code =? 0) then (prim utf8_encode "rejected: missing config key" : Bytes) else (prim utf8_encode "invalid request" : Bytes))))))
}

definition{
  name: preflightAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 5) then (verdictText (verdictCode (Kernel.Wire::fieldAt value 1 1) (Kernel.Wire::fieldAt value 2 1) (Kernel.Wire::fieldAt value 3 1) (Kernel.Wire::fieldAt value 4 1))) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
