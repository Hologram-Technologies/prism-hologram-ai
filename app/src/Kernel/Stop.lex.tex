\begin{lexlean}{Kernel.Stop}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Kernel.Wire}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: stopIdAt,
  parameters: [value : Bytes, position : Nat],
  result: Nat,
  body: (Kernel.Wire::fieldAt value (14 + (prim multiply 4 position : Nat)) 4),
  axioms: [Quot.sound, propext]
}

definition{
  name: stopDiffers,
  parameters: [token : Nat, listed : Nat],
  result: Bool,
  body: ((listed <? token) || (token <? listed))
}

definition{
  name: stopNoneFrom,
  parameters: [value : Bytes, position : Nat, remaining : Nat, token : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => true
    | Nat.succ rest => ((stopDiffers token (stopIdAt value position)) && (stopNoneFrom value (position + 1) rest token))),
  axioms: [Quot.sound, propext],
  recursive_argument: remaining
}

definition{
  name: stopSpent,
  parameters: [generated : Nat, budget : Nat],
  result: Bool,
  body: (budget <=? generated)
}

definition{
  name: stopAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if (((Foundation.Bytes::byteLength value) =? (14 + (prim multiply 4 (Kernel.Wire::fieldAt value 1 1) : Nat))) && ((Kernel.Wire::fieldAt value 1 1) <=? 16)) then (if (stopNoneFrom value 0 (Kernel.Wire::fieldAt value 1 1) (Kernel.Wire::fieldAt value 2 4)) then (if (stopSpent (Kernel.Wire::fieldAt value 6 4) (Kernel.Wire::fieldAt value 10 4)) then (prim utf8_encode "stop: budget" : Bytes) else (prim utf8_encode "continue" : Bytes)) else (prim utf8_encode "stop: end token" : Bytes)) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
