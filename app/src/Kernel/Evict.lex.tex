\begin{lexlean}{Kernel.Evict}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Kernel.Wire}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: evictShort,
  parameters: [needed : Nat, free : Nat, freed : Nat],
  result: Bool,
  body: ((free + freed) <? needed)
}

definition{
  name: evictCode,
  parameters: [pinned : Nat, remaining : Nat, needed : Nat, free : Nat, freed : Nat],
  result: Nat,
  body: (if (evictShort needed free freed) then (if (remaining =? 0) then 0 else (if (pinned =? 0) then 2 else 1)) else 3)
}

definition{
  name: evictText,
  parameters: [code : Nat],
  result: Bytes,
  body: (if (code =? 3) then (prim utf8_encode "enough" : Bytes) else (if (code =? 2) then (prim utf8_encode "evict" : Bytes) else (if (code =? 1) then (prim utf8_encode "keep" : Bytes) else (prim utf8_encode "does not fit" : Bytes))))
}

definition{
  name: evictAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if (((Foundation.Bytes::byteLength value) =? 30) && ((Kernel.Wire::fieldAt value 1 1) <=? 1)) then (evictText (evictCode (Kernel.Wire::fieldAt value 1 1) (Kernel.Wire::fieldAt value 2 4) (Kernel.Wire::fieldAt value 6 8) (Kernel.Wire::fieldAt value 14 8) (Kernel.Wire::fieldAt value 22 8))) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
