\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: exactFits,
  parameters: [activationBits : Nat, weightBits : Nat, groupBits : Nat, accumulatorBits : Nat],
  result: Bool,
  body: (((1 <=? activationBits) && (1 <=? weightBits)) && (((activationBits + weightBits) + groupBits) <=? accumulatorBits))
}

definition{
  name: exactAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 5) then (if (exactFits (Kernel.Wire::fieldAt value 1 1) (Kernel.Wire::fieldAt value 2 1) (Kernel.Wire::fieldAt value 3 1) (Kernel.Wire::fieldAt value 4 1)) then (prim utf8_encode "exact" : Bytes) else (prim utf8_encode "may overflow" : Bytes)) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
