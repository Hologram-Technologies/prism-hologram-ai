\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: trimFits,
  parameters: [context : Nat, fixed : Nat, total : Nat, dropped : Nat],
  result: Bool,
  body: ((fixed <=? context) && ((prim subtract total dropped : Nat) <=? (prim subtract context fixed : Nat)))
}

definition{
  name: trimMinimal,
  parameters: [context : Nat, fixed : Nat, total : Nat, dropped : Nat, last : Nat],
  result: Bool,
  body: ((dropped =? 0) || ((last <=? dropped) && ((prim subtract (prim subtract context fixed : Nat) (prim subtract total dropped : Nat) : Nat) <? last)))
}

definition{
  name: trimmed,
  parameters: [context : Nat, fixed : Nat, total : Nat, dropped : Nat, last : Nat],
  result: Bool,
  body: ((dropped <=? total) && ((trimFits context fixed total dropped) && (trimMinimal context fixed total dropped last)))
}
