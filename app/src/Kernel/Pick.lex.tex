\begin{semanticmodule}
definition{
  name: scoreAt,
  parameters: [value : Bytes, position : Nat],
  result: Nat,
  body: (Kernel.Wire::fieldAt value (7 + (prim multiply 4 position : Nat)) 4),
  axioms: [Quot.sound, propext]
}

definition{
  name: standing,
  parameters: [position : Nat, index : Nat, best : Nat, score : Nat],
  result: Bool,
  body: (((position <? index) && (score <? best)) || (((position =? index) && (score =? best)) || ((index <? position) && (score <=? best))))
}

definition{
  name: holdsFrom,
  parameters: [value : Bytes, position : Nat, remaining : Nat, index : Nat, best : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => true
    | Nat.succ rest => ((standing position index best (scoreAt value position)) && (holdsFrom value (position + 1) rest index best))),
  axioms: [Quot.sound, propext],
  recursive_argument: remaining
}

  axioms: [Quot.sound, propext]
}
