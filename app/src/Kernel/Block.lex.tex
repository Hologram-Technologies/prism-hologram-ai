\begin{lexlean}{Kernel.Block}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Kernel.Wire}
\title{Bounded byte buffer}
\begin{semanticmodule}
definition{
  name: weightCode,
  parameters: [value : Bytes, place : Nat],
  result: Nat,
  body: (prim remainder (prim quotient (Kernel.Wire::fieldAt value (1 + (prim quotient place 2 0 : Nat)) 1) (if ((prim remainder place 2 0 : Nat) =? 0) then 1 else 16) 0 : Nat) 16 0 : Nat)
}

definition{
  name: activationByte,
  parameters: [value : Bytes, place : Nat],
  result: Nat,
  body: (Kernel.Wire::fieldAt value (17 + place) 1)
}

definition{
  name: weightSize,
  parameters: [code : Nat],
  result: Nat,
  body: (if (8 <=? code) then (prim subtract code 8 : Nat) else (prim subtract 8 code : Nat))
}

definition{
  name: activationSize,
  parameters: [byte : Nat],
  result: Nat,
  body: (if (byte <? 128) then byte else (prim subtract 256 byte : Nat))
}

definition{
  name: agrees,
  parameters: [code : Nat, byte : Nat],
  result: Bool,
  body: (if (8 <=? code) then (byte <? 128) else (128 <=? byte))
}

definition{
  name: term,
  parameters: [code : Nat, byte : Nat],
  result: Nat,
  body: (prim multiply (weightSize code) (activationSize byte) : Nat)
}

definition{
  name: positiveFrom,
  parameters: [value : Bytes, place : Nat, remaining : Nat],
  result: Nat,
  body: (match remaining
    | Nat.zero => 0
    | Nat.succ rest => ((if (agrees (weightCode value place) (activationByte value place)) then (term (weightCode value place) (activationByte value place)) else 0) + (positiveFrom value (place + 1) rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: remaining
}

definition{
  name: negativeFrom,
  parameters: [value : Bytes, place : Nat, remaining : Nat],
  result: Nat,
  body: (match remaining
    | Nat.zero => 0
    | Nat.succ rest => ((if (agrees (weightCode value place) (activationByte value place)) then 0 else (term (weightCode value place) (activationByte value place))) + (negativeFrom value (place + 1) rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: remaining
}

definition{
  name: accepts,
  parameters: [claimed : Nat, positive : Nat, negative : Nat],
  result: Bool,
  body: ((claimed + negative) =? (positive + 32768))
}

definition{
  name: blockAnswer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength value) =? 53) then (if (accepts (Kernel.Wire::fieldAt value 49 4) (positiveFrom value 0 32) (negativeFrom value 0 32)) then (prim utf8_encode "exact" : Bytes) else (prim utf8_encode "differs" : Bytes)) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
